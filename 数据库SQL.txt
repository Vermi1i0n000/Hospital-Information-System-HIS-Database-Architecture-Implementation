 1. —— 创建 Schema

/*-------------------------------------
 2. —— 统一的登录用户表
---------------------------------------*/
CREATE TABLE dbo.[User] (
  user_id     INT            IDENTITY(1,1) PRIMARY KEY,
  login_name  NVARCHAR(128)  NOT NULL UNIQUE,
  pwd_hash    VARBINARY(64)  NOT NULL,
  role_code   NVARCHAR(20)   NOT NULL
    CONSTRAINT CK_User_Role
      CHECK (role_code IN (N'Admin', N'Doctor', N'Patient')),
  entity_id   INT            NOT NULL
);
GO

/*-------------------------------------
 3. —— 各业务表按 Schema 创建或迁移
      （这里只示例 Doctor、Patient、Department、Schedule，
       其它表同理，可参照迁移到相应 Schema）
---------------------------------------*/
-- 3.1 科室表到 admin.Department
CREATE TABLE admin.Department (
  dept_id     INT           PRIMARY KEY,
  dept_name   NVARCHAR(50)  NOT NULL,
  location    NVARCHAR(100),
  phone       NVARCHAR(20),
  description NVARCHAR(MAX)
);

-- 3.2 医生表到 doc.Doctor
CREATE TABLE doc.Doctor (
  doctor_id INT            PRIMARY KEY,
  name      NVARCHAR(50)   NOT NULL,
  gender    NVARCHAR(1)    NOT NULL
     CONSTRAINT CK_Doctor_Gender CHECK (gender IN (N'男', N'女')),
  title     NVARCHAR(20)   NOT NULL
     CONSTRAINT CK_Doctor_Title  CHECK (title IN (N'主任医师',N'副主任医师',N'主治医师',N'住院医师')),
  contact   NVARCHAR(20),
  -- password 字段移除，改从 dbo.[User] 验证
  dept_id   INT            NOT NULL
     CONSTRAINT FK_Doctor_Department FOREIGN KEY REFERENCES admin.Department(dept_id)
);

-- 3.3 病人表到 pat.Patient
CREATE TABLE pat.Patient (
  patient_id INT            PRIMARY KEY,
  name       NVARCHAR(50)   NOT NULL,
  gender     NVARCHAR(1)    NOT NULL
     CONSTRAINT CK_Patient_Gender CHECK (gender IN (N'男', N'女')),
  age        INT,
  id_card    NVARCHAR(18)   UNIQUE NOT NULL,
  contact    NVARCHAR(20),
  address    NVARCHAR(100)
  -- password 字段移除，从 dbo.[User] 验证
);

-- 3.4 排班表到 doc.Schedule
CREATE TABLE doc.Schedule (
  schedule_id INT            PRIMARY KEY,
  doctor_id   INT            NOT NULL
     CONSTRAINT FK_Schedule_Doctor FOREIGN KEY REFERENCES doc.Doctor(doctor_id),
  date        DATE           NOT NULL,
  time_slot   NVARCHAR(50)   NOT NULL,
  location    NVARCHAR(10)   NOT NULL
     CONSTRAINT CK_Schedule_Loc  CHECK (location IN (N'门诊', N'住院部')),
  CONSTRAINT UC_Schedule UNIQUE (doctor_id, date, time_slot)
);
-- 3.5 管理员到 admin.Administrator
CREATE TABLE admin.Administrator (
    AdminID INT PRIMARY KEY,
    AdminName NVARCHAR(50)NOT NULL,
    Password NVARCHAR(50)NOT NULL,
	contact VARCHAR(20) 
);
-- 3.6 病房到 admin.Ward
CREATE TABLE admin.Ward (
     ward_id INT  PRIMARY KEY,
    dept_id INT,
    location NVARCHAR(100),
    charge_standard DECIMAL(10,2),
    FOREIGN KEY (dept_id) REFERENCES admin.Department(dept_id)
);
-- 3.7 病床到 admin.Bed
CREATE TABLE admin.Bed (
   ward_id INT,
    bed_no INT,
    status NVARCHAR(4) NOT NULL CHECK (status IN ('空闲', '已占用')),
    PRIMARY KEY (ward_id, bed_no),
    FOREIGN KEY (ward_id) REFERENCES admin.Ward(ward_id)
);
-- 3.8 药品到 doc.Medicine
CREATE TABLE doc.Medicine (
     medicine_id INT  PRIMARY KEY,
    name NVARCHAR(100) NOT NULL,
    specification NVARCHAR(50),
    price DECIMAL(10,2) NOT NULL,
    stock INT NOT NULL CHECK (stock >= 0),
    manufacturer NVARCHAR(100),
    description NVARCHAR(MAX)
);
-- 3.15 挂号到pat.Registration
CREATE TABLE pat.Registration (
  reg_id INT  PRIMARY KEY,
    patient_id INT NOT NULL,
    dept_id INT NOT NULL,
    doctor_id INT NOT NULL,
    reg_time DATETIME NOT NULL,
    fee DECIMAL(10,2) NOT NULL,
    status NVARCHAR(10) CHECK (status IN ('待就诊', '已就诊')),
    FOREIGN KEY (patient_id) REFERENCES pat.Patient(patient_id),
    FOREIGN KEY (dept_id) REFERENCES admin.Department(dept_id),
    FOREIGN KEY (doctor_id) REFERENCES doc.Doctor(doctor_id)
);

-- 3.9 处方到 doc.Prescription
CREATE TABLE doc.Prescription (
    prescription_id INT  PRIMARY KEY,
    reg_id INT NOT NULL UNIQUE,
    diagnosis NVARCHAR(MAX) NOT NULL,
    date DATE NOT NULL,
    FOREIGN KEY (reg_id) REFERENCES pat.Registration(reg_id)
);
-- 3.10药品清单到 doc.Prescription
CREATE TABLE doc.Medicine_List (
    prescription_id INT NOT NULL,
    medicine_id INT NOT NULL,
    quantity INT NOT NULL CHECK (quantity > 0),
    usage NVARCHAR(MAX),
    dosage NVARCHAR(50),
    PRIMARY KEY (prescription_id, medicine_id),
    FOREIGN KEY (prescription_id) REFERENCES doc.Prescription(prescription_id),
    FOREIGN KEY (medicine_id) REFERENCES doc.Medicine(medicine_id)
);
-- 3.11处方费用到 doc.Prescription_Fee
CREATE TABLE doc.Prescription_Fee (
   prescription_id INT PRIMARY KEY,
    diagnosis_fee DECIMAL(10,2) NOT NULL,
    medicine_fee DECIMAL(10,2) NOT NULL,
    total_fee DECIMAL(10,2) NOT NULL,
    payment_status NVARCHAR(10) CHECK (payment_status IN ('未支付', '已支付')),
    FOREIGN KEY (prescription_id) REFERENCES doc.Prescription(prescription_id)
);
-- 3.12住院档案到 pat.Hospitalization
CREATE TABLE pat.Hospitalization (
   hospitalization_id INT PRIMARY KEY,
    patient_id INT NOT NULL,
    admission_time DATETIME NOT NULL,
    discharge_time DATETIME,
    ward_id INT NOT NULL,
    bed_no INT NOT NULL,
    doctor_id INT NOT NULL,
    status NVARCHAR(10) CHECK (status IN ('在院', '已出院')),
    FOREIGN KEY (patient_id) REFERENCES pat.Patient(patient_id),
    FOREIGN KEY (ward_id, bed_no) REFERENCES admin.Bed(ward_id, bed_no),
    FOREIGN KEY (doctor_id) REFERENCES doc.Doctor(doctor_id)
);
-- 3.13住院记录到 pat.Hospitalization_Record
CREATE TABLE pat.Hospitalization_Record (
    record_id INT  PRIMARY KEY,
    hospitalization_id INT NOT NULL,
    record_time DATETIME NOT NULL,
    condition_description NVARCHAR(MAX) NOT NULL,
    treatment_plan NVARCHAR(MAX) NOT NULL,
    FOREIGN KEY (hospitalization_id) REFERENCES pat.Hospitalization(hospitalization_id)
);
-- 3.14住院费用到 pat.Hospitalization_Fee
CREATE TABLE pat.Hospitalization_Fee (
        fee_id INT PRIMARY KEY,
    hospitalization_id INT NOT NULL,
    fee_date DATE NOT NULL,
    bed_fee DECIMAL(10,2) NOT NULL,
    treatment_fee DECIMAL(10,2) NOT NULL,
    medicine_fee DECIMAL(10,2) NOT NULL,
    other_fee DECIMAL(10,2) NOT NULL,
    total_fee DECIMAL(10,2) NOT NULL,
    FOREIGN KEY (hospitalization_id) REFERENCES pat.Hospitalization(hospitalization_id)
);


GO

/*-------------------------------------
 6. —— 登录存储过程 ucsp_Login
---------------------------------------*/
CREATE PROCEDURE dbo.usp_Login
  @login_name NVARCHAR(128),
  @pwd_plain  NVARCHAR(100)
AS
BEGIN
  SET NOCOUNT ON;
  DECLARE @hash   VARBINARY(64),
          @role   NVARCHAR(20),
          @eid    INT;
  SET @hash = HASHBYTES('SHA2_256', @pwd_plain);

  SELECT @role=role_code, @eid=entity_id
    FROM dbo.[User]
   WHERE login_name=@login_name
     AND pwd_hash=@hash;

  IF @role IS NULL
  BEGIN
    RAISERROR('用户名或密码错误。',16,1);
    RETURN;
  END

  -- 返回登录结果
  SELECT @role AS UserRole, @eid AS EntityID;
END;
GO

 7. —— 创建功能视图
---------------------------------------*/
-- 7.1 管理员查询所有科室、医生、病房、药品等（AdminRole 已全权）
--    （可由应用层直接查询 admin.Department、doc.Doctor、…）

-- 7.2 医生端：查看当日排班
CREATE PROCEDURE doc.GetDoctorTodaySchedule
    @DocID INT
AS
BEGIN
    SELECT 
        s.schedule_id,
        s.date,
        s.time_slot,
        s.location,
        d.dept_name
    FROM doc.Schedule AS s
    JOIN doc.Doctor AS doc ON s.doctor_id = doc.doctor_id
    JOIN admin.Department AS d ON doc.dept_id = d.dept_id
    WHERE s.doctor_id = @DocID AND s.date = CAST(GETDATE() AS DATE);
END;
GO


-- 7.3 医生端：查看自己接诊病人列表
CREATE PROCEDURE doc.GetMyPatients
    @DocID INT
AS
BEGIN
    SELECT 
        r.reg_id,
        r.reg_time,
        r.status,
        p.name AS patient_name,
        d.dept_name,
        s.time_slot
    FROM pat.Registration r
    JOIN pat.Patient p ON p.patient_id = r.patient_id
    JOIN doc.Schedule s ON r.doctor_id = s.doctor_id AND CAST(r.reg_time AS DATE) = s.date
    JOIN admin.Department d ON r.dept_id = d.dept_id
    WHERE r.doctor_id = @DocID;
END;
GO


-- 7.4 病人端：查看自己的挂号
CREATE VIEW pat.MyRegistrations AS
SELECT 
    r.reg_id, 
    r.dept_id, 
    dep.dept_name, 
    r.reg_time, 
    r.status
FROM pat.Registration AS r
JOIN admin.Department AS dep ON r.dept_id = dep.dept_id
WHERE r.patient_id = (
    SELECT entity_id 
    FROM dbo.[User] 
    WHERE login_name = SUSER_SNAME()
);
GO


-- 7.5 病人端：查看自己的处方
CREATE VIEW pat.MyPrescriptions AS
SELECT pr.prescription_id, pr.date, pr.diagnosis
FROM doc.Prescription AS pr
JOIN pat.Registration AS r ON pr.reg_id = r.reg_id
WHERE r.patient_id = (
    SELECT entity_id FROM dbo.[User] WHERE login_name = SUSER_SNAME()
);
GO


/*-------------------------------------
 8. —— 统计视图（rpt Schema）
---------------------------------------*/
-- 8.1 科室排班统计
CREATE VIEW rpt.DeptScheduleStats AS
SELECT d.dept_name, s.date, COUNT(*) AS 排班数量
FROM doc.Schedule AS s
JOIN doc.Doctor AS doc ON s.doctor_id = doc.doctor_id
JOIN admin.Department AS d ON doc.dept_id = d.dept_id
GROUP BY d.dept_name, s.date;
GO


-- 8.2 医生工作量统计
CREATE VIEW rpt.DoctorWorkloadStats AS
SELECT doc.doctor_id, doc.name AS 医生姓名, COUNT(r.reg_id) AS 接诊总数
FROM doc.Doctor AS doc
LEFT JOIN pat.Registration AS r ON doc.doctor_id = r.doctor_id
GROUP BY doc.doctor_id, doc.name;
GO


-- 8.3 病人治疗情况统计
CREATE VIEW rpt.PatientTreatmentStats AS
SELECT p.name   AS 病人姓名,
       COUNT(r.reg_id)     AS 挂号次数,
       COUNT(pr.prescription_id) AS 处方次数
FROM pat.Patient AS p
LEFT JOIN pat.Registration AS r ON p.patient_id = r.patient_id
LEFT JOIN doc.Prescription AS pr ON pr.reg_id = r.reg_id
GROUP BY p.name;
GO
